Q1. Check each character of the string. If it is a lower case then subtract 32 from its ASCII value to convert it into uppercase.

CODE

#include <stdio.h>
void main()
{
    char str[20];
    int i;

    printf("Enter a string: ");
    gets(str);

    for(i = 0; str[i] != '\0'; i++)
    {
        if(str[i] >= 'a' && str[i] <= 'z')

}

QUESTION 18

Q18.Write a program to enter the string from user and check string palindrome example (using reverse).

CODE

#include <stdio.h>
#include <string.h>
void main()
{
    char str[20];
    int i, num, check = 1;

    printf("Enter a string: ");
    gets(str);

    num = strlen(str) - 1;

    for(i = 0; i < num; i++)
    {
        if(str[i] != str[num])
        {
            check = 0;
            break;
        }
        num--;
    }

    if(check == 1)
      printf("String is palindrome.");
    else
      printf("String is not palindrome.");
}

OUTPUT

C:\TURBOC3\BIN>TC
Enter a string: level
String is palindrome.
