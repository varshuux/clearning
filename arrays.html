QUESTION 1

Q1. Write a program to enter an array from user and display all the values.

CODE

#include <stdio.h>
#include <dos.h>
void main()
{
    int num[5], i;

    printf("Enter 5 values: \n");

    for(i = 0; i <= 4; i++)
    {
        scanf("%d",&num[i]);
    }

    printf("Array values are :\n");

    for(i = 0; i <= 4; i++)
    {
        printf("%d ",num[i]);
    }
    delay(3000);
}

CODE

#include <dos.h>

QUESTION 10

Q10. Write a program to enter 5 values of array from user, enter a value from user and search this value in array. If value is found then print "Found" otherwise print "Not Found".

CODE

#include <stdio.h>
#include <dos.h>
void main()
{
    int num[5], find = 0, search, i;

    printf("Enter 5 values: \n");

    for(i = 0; i <= 4; i++)
    {
        scanf("%d",&num[i]);
    }

    printf("Enter value to search: ");
    scanf("%d",&search);

    for(i = 0; i <= 4; i++)
    {
        if(num[i] == search)
        {
            find = 1;
            break;
        }
    }

    if(find == 0)
        printf("Not Found");
    else
        printf("Found");

    delay(3000);
}

OUTPUT

C:\TURBOC3\BIN>TC
Enter 5 values:
15
28
39
47
62
Enter value to search: 39
Found

QUESTION 11

Q11. Write a program to enter five values of array and sort them in ascending order.

CODE

#include <stdio.h>
#include <dos.h>
void main()
{
    int num[5], i, j, temp;

    printf("Enter 5 values: \n");

    for(i = 0; i <= 4; i++)
    {
        scanf("%d",&num[i]);
    }

    for(i = 0; i <= 3; i++)
    {
        for(j = 0; j <= 3-i; j++)
        {
            if(num[j] > num[j+1])
            {
                temp = num[j];
                num[j] = num[j+1];
                num[j+1] = temp;
            }
        }
    }

    printf("Array in ascending order:\n");

    for(i = 0; i <= 4; i++)
    {
        printf("%d ",num[i]);
    }

    delay(3000);
}

OUTPUT

C:\TURBOC3\BIN>TC
Enter 5 values:
72
18
54
26
41
Array in ascending order:
18 26 41 54 72

QUESTION 12

Q12. Write a program to find the maximum value of an array.

CODE

#include <stdio.h>
#include <dos.h>
void main()
{
    int num[5], i, max;

    printf("Enter 5 values: \n");

    for(i = 0; i <= 4; i++)
    {
        scanf("%d",&num[i]);
    }

    max = num[0];

    for(i = 1; i <= 4; i++)
    {
        if(num[i] > max)
            max = num[i];
    }

    printf("Maximum value = %d",max);

    delay(3000);
}

76
21
59
