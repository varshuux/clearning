Q1.Write a program print the pattern on the screen i.e. ? 11111 11111 11111

CODE

#include <stdio.h>
#include <conio.h>
#include <dos.h>
void main()
{
int i, j;

    for(i = 0; i <= 3; i++)
    {
        for(j = 0; j <= 5; j++)
        {
            printf("5");
        }
      printf("\n");
    }
  delay(3000);
}
