QUESTION 1

Q1.Create structure of employee which there is emp.ID, name, designation of 50 employees from user, print details in tabular form.

CODE

#include <stdio.h>
#include <dos.h>
void main()
{
    struct emp
{
    int id;
    char name[20];
    char designation[30];
};

void main()
{
    struct emp e[50];
    int i;

    for(i = 0; i < 50; i++)
    {
        printf("\nEnter details of employee %d\n", i + 1);

        printf("Enter Employee Id: ");
        scanf("%d",&e[i].id);

        printf("Enter Name: ");
        scanf("%s",e[i].name);

        printf("Enter Designation: ");
        scanf("%s",e[i].designation);
    }

    printf("\n\nEMPLOYEE DETAILS\n");
    printf("-----------------------------\n");
    printf("ID\tNAME\t\tDESIGNATION\n");
    printf("-----------------------------\n");

    for(i = 0; i < 50; i++)
    {
      printf("%d\t%-15s\t%s\n", e[i].id, e[i].name, e[i].designation);
    }
  delay(3000);
}

QUESTION 4

Q4.Enter student details like roll name, fee, DOB and print all the detail by ascending order of roll number.

CODE

#include <stdio.h>
#include <dos.h>
void main()
{
    struct stu
{
    int roll;
    char name[20];
    float fee;
    char dob[15];
};

void main()
{
    struct stu s[50], temp;
    int i, j, n;

    printf("Enter number of student(50 max): ");
    scanf("%d",&n);

    for(i = 0; i < n; i++)
    {
        printf("\nEnter details of student %d\n", i + 1);

        printf("Enter Roll Number: ");
        scanf("%d",&s[i].roll);
        fflush(stdin);

        printf("Enter Name: ");
        gets(s[i].name);

        printf("Enter Fee: ");
        scanf("%f",&s[i].fee);
        fflush(stdin);

        printf("Enter DOB (DD-MM-YYYY): ");
        gets(s[i].dob);
    }

    for(i = 0; i < n - 1; i++)
    {
        for(j = i + 1; j < n; j++)
        {
            if(s[i].roll > s[j].roll)
            {
                temp = s[i];
                s[i] = s[j];
                s[j] = temp;
            }
        }
    }



    printf("\n\nSTUDENT DETAILS\n");
    printf("---------------------------------------------------\n");
    printf("Roll\tNAME\t\tFEE\t\tDOB\n");
    printf("---------------------------------------------------\n");

    for(i = 0; i < n; i++)
    {
      printf("%d\t%-15s\t%.2f\t%s\n", s[i].roll, s[i].name, s[i].fee, s[i].dob);
    }
        delay(3000);
}
