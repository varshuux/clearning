void input()

void fact()
{
	int i, fact = 1;
	for(i = 1; i <= num; i++)
	  fact = fact * i;

	printf("Factorial = %d\n",fact);
}

void reverse()
{
	int n, rem, rev = 0;
	n = num;

	while(n > 0)
	{
		rem = n % 10;
		rev = rev * 10 + rem;
		n = n / 10;
	}

	printf("Reverse = %d\n", rev);
}

void palindrome()
{
	int n, rem, rev = 0;
	n = num;

	while(n > 0)
	{
		rem = n % 10;
		rev = rev * 10 + rem;
		n = n / 10;
	}

	if(num == rev)
	  printf("Palindrome Number\n");
	else
	  printf("Not a Palindrome Number\n");
}

void prime()
{
	int i, count = 0;

	for(i = 1; i <= num; i++)
	{
		if(num % i == 0)
		count++;
	}

	if(count == 2)
	  printf("Prime Number\n");
	else
	  printf("Not a Prime Number\n");
}

void armstrong()
{
	int n, rem, sum = 0;
	n = num;

	while(n > 0)
	{
		rem = n % 10;
		sum = sum + rem * rem * rem;
		n = n /10;
	}

	if(num == sum)
	  printf("Armstrong number\n");
	else
	  printf("Not an Armstrong number\n");
}

{
	int a = 0, b = 1, c, i;

		printf("%d ",a);

		b = c;
	}

Q5.Write a program to swap numbers using Call by address.

CODE

#include <stdio.h>
void swap(int *a, int *b)
{
	int temp;

	temp = *a;
	*a = *b;
	*b = temp;
}

void main()
{
	int x, y;

	printf("Enter first number: ");
	scanf("%d",&x);
	printf("Enter second number: ");
	scanf("%d",&y);

	printf("\nBefore swapping:\n");
	printf("First number = %d\n",x);
	printf("Second number = %d\n",y);

	swap(&x,&y);

	printf("\nAfter swapping:\n");
	printf("First number = %d\n",x);
	printf("Second number = %d\n",y);
}

Q7.Make function for find out power of number, enter the number and power from user.

CODE

#include <stdio.h>
int power(int a, int b)
{
	int i, result = 1;
	for(i = 1; i <= b; i++)
	{
		result = result * a;
	}

	return result;
}

void main()
{
	int base, exp, res;

	printf("Enter base: ");
	scanf("%d",&base);
	printf("Enter exponent: ");
	scanf("%d",&exp);
	
	res = power(base,exp);

	printf("\n%d raised to the power %d = %d",base ,exp ,res);
}

OUTPUT

Enter exponent: 5
