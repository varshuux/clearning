QUESTION 4

Q4. Write programs using local and global variables to find the factorial, reverse, palindrome, prime number, Armstrong number, and Fibonacci series of a given number.

CODE

#include <stdio.h>
#include <conio.h>
#include <dos.h>

int num;

void input()
{
    printf("Enter a number: ");
    scanf("%d",&num);
}

void fact()
{
    int i,fact=1;

    if(num < 0)
    {
        printf("Factorial Not Possible\n");
        return;
    }

    for(i = 1; i <= num; i++)
        fact = fact * i;

    printf("Factorial = %d\n",fact);
}

void reverse()
{
    int n,rem,rev=0;

    n = num;

    while(n > 0)
    {
        rem = n % 10;
        rev = rev * 10 + rem;
        n = n / 10;
    }

    printf("Reverse = %d\n",rev);
}

void palindrome()
{
    int n,rem,rev=0;

    n = num;

    while(n > 0)
    {
        rem = n % 10;
        rev = rev * 10 + rem;
        n = n / 10;
    }

    if(num == rev)
        printf("Palindrome Number\n");
    else
        printf("Not a Palindrome Number\n");
}

void prime()
{
    int i,count=0;

    if(num <= 1)
    {
        printf("Not a Prime Number\n");
        return;
    }

    for(i = 1; i <= num; i++)
    {
        if(num % i == 0)
            count++;
    }

    if(count == 2)
        printf("Prime Number\n");
    else
        printf("Not a Prime Number\n");
}

void armstrong()
{
    int n,rem,sum=0;

    n = num;

    while(n > 0)
    {
        rem = n % 10;
        sum = sum + rem * rem * rem;
        n = n / 10;
    }

    if(num == sum && num >= 0)
        printf("Armstrong Number\n");
    else
        printf("Not an Armstrong Number\n");
}

void fibonacci()
{
    int a=0,b=1,c,i;

    if(num <= 0)
    {
        printf("Invalid Number of Terms\n");
        return;
    }

    printf("Fibonacci Series: ");

    for(i = 1; i <= num; i++)
    {
        printf("%d ",a);

        c = a + b;
        a = b;
        b = c;
    }

    printf("\n");
}

void main()
{
    clrscr();

    input();
    fact();

    input();
    reverse();

    input();
    palindrome();

    input();
    prime();

    input();
    armstrong();

    input();
    fibonacci();

    delay(3000);
}

QUESTION 5

Q5. Write a program to swap two numbers using the call by address method.

CODE

#include <stdio.h>
#include <conio.h>
#include <dos.h>

void swap(int *a,int *b)
{
    int temp;

    temp = *a;
    *a = *b;
    *b = temp;
}

void main()
{
    int x,y;

    clrscr();

    printf("Enter first number: ");
    scanf("%d",&x);

    printf("Enter second number: ");
    scanf("%d",&y);

    printf("\nBefore swapping:\n");
    printf("First number = %d\n",x);
    printf("Second number = %d\n",y);

    swap(&x,&y);

    printf("\nAfter swapping:\n");
    printf("First number = %d\n",x);
    printf("Second number = %d\n",y);

    delay(3000);
}

OUTPUT

Enter first number: 25
Enter second number: 50

Before swapping:
First number = 25
Second number = 50

After swapping:
First number = 50
Second number = 25

QUESTION 6

Q6. Write a program to swap two numbers using the call by value method.

CODE

#include <stdio.h>
#include <conio.h>
#include <dos.h>

void swap(int a,int b)
{
    int temp;

    temp = a;
    a = b;
    b = temp;

    printf("\nAfter swapping inside function:\n");
    printf("First number = %d\n",a);
    printf("Second number = %d\n",b);
}

void main()
{
    int x,y;

    clrscr();

    printf("Enter first number: ");
    scanf("%d",&x);

    printf("Enter second number: ");
    scanf("%d",&y);

    printf("\nBefore swapping:\n");
    printf("First number = %d\n",x);
    printf("Second number = %d\n",y);

    swap(x,y);

    printf("\nAfter function call:\n");
    printf("First number = %d\n",x);
    printf("Second number = %d\n",y);

    delay(3000);
}

OUTPUT

Enter second number: 17

Before swapping:
First number = 87
Second number = 17

After swapping inside function:
First number = 17
Second number = 87

QUESTION 7

Q7. Write a program using a function to calculate the power of a number. Enter the base and exponent from the user.

CODE

#include <stdio.h>
#include <conio.h>
#include <dos.h>

int power(int a,int b)
{
    int i,result=1;

    for(i = 1; i <= b; i++)
    {
        result = result * a;
    }

    return result;
}

void main()
{
    int base,exp,res;

    clrscr();

    printf("Enter base: ");
    scanf("%d",&base);

    printf("Enter exponent: ");
    scanf("%d",&exp);

    if(exp < 0)
    {
        printf("Negative exponent is not supported");
    }
    else
    {
        res = power(base,exp);

        printf("\n%d raised to the power %d = %d",
               base,exp,res);
    }

    delay(3000);
}

OUTPUT

Enter base: 2
Enter exponent: 5

2 raised to the power 5 = 32
