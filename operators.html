QUESTION 10

Q10.Write a program to enter a lowercase alphabet and convert it into uppercase..

CODE

#include <stdio.h>
#include <conio.h>
void main()
{
char ch;

    do{
        printf("Enter a lowercase alphabet: ");
        ch = getch();
        printf("\n");
    }   while(ch < 'a' || ch > 'z');
    ch = ch - 32;
    printf("Uppercase alphabet = %c",ch);

    return 0;
}

OUTPUT

C:\TURBOC3\BIN>TC
Enter a lowercase alphabet: b

QUESTION 11

Q11.Write a program to enter marks of three subjects and calculate their average.

CODE

#include <stdio.h>
#include <conio.h>
void main()
{
int num1,num2,num3;
    float average;

    printf("Enter first student marks: ");
    scanf("%d",&num1);
    printf("Enter second student marks: ");
    scanf("%d",&num2);
    printf("Enter third student marks: ");
    scanf("%d",&num3);

    average = (num1 + num2 + num3) / 3.0;

    printf("Average marks = %.2f",average);

    return 0;
}

OUTPUT

C:\TURBOC3\BIN>TC
Enter first student marks: 70
Enter second student marks: 80
Enter third student marks: 90
Average marks = 80

QUESTION 12

Q12.Write a program to calculate interest using principal, rate and time.

CODE

#include <stdio.h>
#include <conio.h>
void main()
{
float p,r,t,si;
    
    printf("Enter principal amount: ");
    scanf("%f",&p);
    printf("Enter rate of interest: ");
    scanf("%f",&r);
    printf("Enter time in years: ");
    scanf("%f",&t);

    si = (p * r * t) / 100;

    printf("Simple Interest = %.2f",si);
    return 0;
}

OUTPUT

C:\TURBOC3\BIN>TC
Enter principal amount: 5000
Enter rate of interest: 5
Enter time in years: 2
Simple Interest = 500.00
