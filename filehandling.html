QUESTION 1

Q1. Enter file name and data from user and put data in file but data should be in capital letter.

CODE

#include <stdio.h>
#include <stdio.h>

void main()
{
    char f[10], data[30];
    int i;
    File * New;

    printf("Enter file name: ");
    gets(f);

    printf("Enter the data: ");
    gets(data);

    New = fopen(f,"w");

    for(i = 0; data[i] != '\0'; i++)
    {
        if(data[i] >= 'a' && data[i] <= 'z')
            data[i] = data[i] - 32;
    }

    fprintf(New, "%s", data);
    fclose(New);

    printf("\nData entered successfully!");
}

QUESTION 6

Q6. Create a file and count the number of characters in it.

CODE

#include <stdio.h>
#include <conio.h>

void main()
{
    char f[10], data[30], a;
    int x, count = 0;
    FILE * New;

    printf("Enter file name: ");
    gets(f);

    printf("First enter then use CTRL + S to save... \nFor next line press enter 2 times...\nEnter the data: ");

    New = fopen(f,"w");

    do
    {
        gets(data);
        fprintf(New, "%s\n", data);
        a = getch();
    }
    while(a != 19);

    printf("\nSaved successfully\n\n");
    fclose(New);

    New = fopen(f,"r");

    do
    {
        x = fgetc(New);

        if(x != -1)
            count++;

    }
    while(x != -1);

    fclose(New);

    printf("File have total = %d characters.",count);
}

OUTPUT


C:\TURBOC3\BIN>TC
Enter file name: f1

Enter the data: Hello world

QUESTION 7

Q7. Create a file and count the number of words in it.

CODE

#include <stdio.h>
#include <conio.h>

void main()
{
    char f[10], data[50], a, *c;
    int x, count = 1;
    FILE * New;

    printf("Enter file name: ");
    gets(f);

    printf("First enter then use CTRL + S to save... \nFor next line press enter 2 times...\nEnter the data: ");

    New = fopen(f,"w");

    do
    {
        gets(data);
        fprintf(New, "%s\n", data);
        a = getch();
    }
    while(a != 19);

    printf("\nSaved successfully\n\n");
    fclose(New);

    New = fopen(f, "r");

    do
    {
        c = fgets(data, 50, New);

        if(c != 0)
        {
            for(x = 0; data[x] != '\0'; x++)
            {
                if(data[x] == ',' || data[x] == ' ')
                {
                    count++;
                    x++;
                }
            }
        }

    }
    while(c != 0);

    fclose(New);

    printf("File have total = %d words.",count);
}

OUTPUT

Enter file name: f1

Enter the data: Hello, jai maharasthra

Saved successfully

File have total = 5 words.
