Q20. Write a program to enter a date, month, and year from the user and check whether the entered date is valid or not.

CODE

#include <dos.h>

{
int d,m,y,maxdays;

    printf("Enter day: ");
    scanf("%d",&d);

    printf("Enter month: ");
    scanf("%d",&m);

    printf("Enter year: ");
    scanf("%d",&y);

    if(m < 1 || m > 12 || d < 1 || y < 1)
        printf("Invalid Date");
    else
    {
        if(m == 4 || m == 6 || m == 9 || m == 11)
          maxdays = 30;
        else if(m == 2)
        {
            if(y % 400 == 0 || (y % 4 == 0 && y % 100 != 0))
              maxdays = 29;
            else
              maxdays = 28;
        }
        else
          maxdays = 31;

        if(d <= maxdays)
          printf("Valid Date");
        else
          printf("Invalid Date");
    }

    delay(5000);
}

QUESTION 21

Q21. Write a program to enter the current date and the date of birth of the user and calculate the user's age in years, months, and days.

CODE

#include <stdio.h>
#include <conio.h>
#include <dos.h>
void main()
{
int d1,m1,y1;
int d2,m2,y2;
int days,months,years;
int mdays,prevmonth,prevyear;

    clrscr();

    printf("Enter current date (day month year): ");
    scanf("%d %d %d",&d1,&m1,&y1);

    printf("Enter date of birth (day month year): ");
    scanf("%d %d %d",&d2,&m2,&y2);

    if(y1 < y2 || (y1 == y2 && m1 < m2) ||
       (y1 == y2 && m1 == m2 && d1 < d2))
    {
        printf("Invalid Dates");
    }
    else
    {
        days = d1 - d2;
        months = m1 - m2;
        years = y1 - y2;

        if(days < 0)
        {
            prevmonth = m1 - 1;
            prevyear = y1;

            if(prevmonth == 0)
            {
                prevmonth = 12;
                prevyear--;
            }

            if(prevmonth == 4 || prevmonth == 6 ||
               prevmonth == 9 || prevmonth == 11)
                mdays = 30;
            else if(prevmonth == 2)
            {
                if(prevyear % 400 == 0 ||
                   (prevyear % 4 == 0 && prevyear % 100 != 0))
                    mdays = 29;
                else
                    mdays = 28;
            }
            else
                mdays = 31;

            days = days + mdays;
            months--;
        }

        if(months < 0)
        {
            months = months + 12;
            years--;
        }

        printf("Age = %d years, %d months, %d days",
               years,months,days);
    }

    delay(5000);
}

OUTPUT

Enter current date (day month year): 28 10 2026
Enter date of birth (day month year): 14 8 2008
Age = 18 years, 2 months, 14 days
