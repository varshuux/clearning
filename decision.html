#include <dos.h>

{

    int d2, m2, y2;

    int mdays;

    clrscr();

    printf("Enter current date (day month year): ");
    scanf("%d %d %d", &d1, &m1, &y1);

    printf("Enter date of birth (day month year): ");
    scanf("%d %d %d", &d2, &m2, &y2);

    if(y1 < y2 || (y1 == y2 && m1 < m2) ||
       (y1 == y2 && m1 == m2 && d1 < d2))
    {
        printf("Invalid dates");
    }
    else
    {
        days = d1 - d2;
        months = m1 - m2;
        years = y1 - y2;

        if(days < 0)
        {
            m1--;

            if(m1 == 4 || m1 == 6 || m1 == 9 || m1 == 11)
                mdays = 30;
            else if(m1 == 2)
            {
                if(y1 % 400 == 0 ||
                   (y1 % 4 == 0 && y1 % 100 != 0))
                    mdays = 29;
                else
                    mdays = 28;
            }
            else
                mdays = 31;

            days = days + mdays;
            months = months - 1;
        }

        if(months < 0)
        {
            months = months + 12;
            years = years - 1;
