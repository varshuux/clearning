QUESTION 2

Q2. Write a program to read a character from keyboard and display it on the screen.

CODE

#include <stdio.h>
void main()
{
    char ch;
    printf("ENTER A CHARACTER: ");
    scanf(" %c", &ch);
    printf("Your character is = %c", ch);
}

OUTPUT

ENTER A CHARACTER: M
Your character is = M

QUESTION 6

Q6. Write a program for printing your name on the output screen but your name should blink 5 times.

CODE

#include <stdio.h>
#include <conio.h>
#include <dos.h>

void main()
{
    int i;
    clrscr();

    for(i = 1; i <= 5; i++)
    {
        printf("VARSHA ");
        delay(500);
        clrscr();
        delay(500);
    }

    getch();
}

OUTPUT

VARSHA

QUESTION 9

Q9. Write a program to enter a password and display it in the form of ******.

CODE

#include <stdio.h>
#include <conio.h>

void main()
{
    char pass[7];
    int i;

    clrscr();

    printf("Enter 6 character Password: ");

    for(i = 0; i < 6; i++)
    {
        pass[i] = getch();
        printf("*");
    }

    pass[6] = '\0';
}

OUTPUT

Enter 6 character Password: ******

QUESTION 10

Q10. Enter two characters from the user and print their ASCII values.

CODE

#include <stdio.h>
#include <conio.h>

void main()
{
    char ch1, ch2;

    clrscr();

    printf("Enter first character: ");
    ch1 = getche();

    printf("\nEnter second character: ");
    ch2 = getche();

    printf("\nASCII value of first character = %d\n", ch1);
    printf("ASCII value of second character = %d\n", ch2);
}

OUTPUT

Enter first character: C
Enter second character: D
ASCII value of first character = 67
ASCII value of second character = 68
