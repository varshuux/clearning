        for(j = 0; j<=1; j++)
        {
            printf("%d ",&num[i][j]);
        }
        printf("\n");

Q3.Write a program to enter 2 matrix from user and add them.

CODE

#include <stdio.h>
void main()
{
    int num[2][2], b[2][2], sum[2][2], i, j;

    printf("Enter 4 values in first matrix: \n");
    
    for(i = 0; i <= 1; i++)
    {
        for(j = 0; j<=1; j++)
        {
            scanf("%d",&num[i][j]);
        }
    }
    printf("Enter 4 values in second matrix: \n");
    
    for(i = 0; i <= 1; i++)
    {
        for(j = 0; j<=1; j++)
        {
            scanf("%d",&b[i][j]);
        }
    }


    for(i = 0; i <= 1; i++)
    {
        for(j = 0; j<=1; j++)
        {
            sum[i][j] = num[i][j] + b[i][j];
        }
    }


    printf("\nAddition of matrices : \n");
    for(i = 0; i <= 1; i++)
    {
        for(j = 0; j<=1; j++)
        {
            printf("%d ",&sum[i][j]);
        }
        printf("\n");
    }   
    return 0;
}
