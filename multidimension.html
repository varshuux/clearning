QUESTION 1

Q1. Write a program to enter four numbers from the user and display them in the form of a 2x2 matrix.

CODE

#include <stdio.h>
void main()
{
    int num[2][2], i, j;

    printf("Enter 4 values: \n");

    for(i = 0; i <= 1; i++)
    {
        for(j = 0; j <= 1; j++)
        {
            scanf("%d",&num[i][j]);
        }
    }

    printf("Matrix is: \n");

    for(i = 0; i <= 1; i++)
    {
        for(j = 0; j <= 1; j++)
        {
            printf("%d ",num[i][j]);
        }
        printf("\n");
    }

    return 0;
}

QUESTION 3

Q3. Write a program to enter two matrices from the user and display their addition.

CODE

#include <stdio.h>
void main()
{
    int num[2][2], b[2][2], sum[2][2], i, j;

    printf("Enter 4 values in first matrix:\n");

    for(i = 0; i <= 1; i++)
    {
        for(j = 0; j <= 1; j++)
        {
            scanf("%d",&num[i][j]);
        }
    }

    printf("Enter 4 values in second matrix:\n");

    for(i = 0; i <= 1; i++)
    {
        for(j = 0; j <= 1; j++)
        {
            scanf("%d",&b[i][j]);
        }
    }

    for(i = 0; i <= 1; i++)
    {
        for(j = 0; j <= 1; j++)
        {
            sum[i][j] = num[i][j] + b[i][j];
        }
    }

    printf("\nAddition of Matrices:\n");

    for(i = 0; i <= 1; i++)
    {
        for(j = 0; j <= 1; j++)
        {
            printf("%d ",sum[i][j]);
        }
        printf("\n");
    }

    return 0;
}

QUESTION 5

Q5. Write a program to enter two matrices from the user and display their multiplication.

CODE

#include <stdio.h>
void main()
{
    int num[2][2], b[2][2], multi[2][2], i, j, k;

    printf("Enter 4 values in first matrix:\n");

    for(i = 0; i <= 1; i++)
    {
        for(j = 0; j <= 1; j++)
        {
            scanf("%d",&num[i][j]);
        }
    }

    printf("Enter 4 values in second matrix:\n");

    for(i = 0; i <= 1; i++)
    {
        for(j = 0; j <= 1; j++)
        {
            scanf("%d",&b[i][j]);
        }
    }

    for(i = 0; i <= 1; i++)
    {
        for(j = 0; j <= 1; j++)
        {
            multi[i][j] = 0;

            for(k = 0; k <= 1; k++)
            {
                multi[i][j] = multi[i][j] + num[i][k] * b[k][j];
            }
        }
    }

    printf("\nMultiplication of Matrices:\n");

    for(i = 0; i <= 1; i++)
    {
        for(j = 0; j <= 1; j++)
        {
            printf("%d ",multi[i][j]);
        }
        printf("\n");
    }

    return 0;
}
